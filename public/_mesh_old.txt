// True 3-tier topology: Internet → Router → (APs) → devices.
// Devices proven to be on an AP's radio hang off that AP; everything else
// hangs off the router.
function renderMeshSvg(d) {
  const svg = $('#meshSvg');
  const W = 1000, H = 660;
  svg.setAttribute('viewBox', `0 0 ${W} ${H}`);
  const internet = { x: W / 2, y: 60 };
  const router = { x: W / 2, y: 190 };
  const pihole = { x: 130, y: 190 };

  const aps = (d.aps || []).filter((a) => a.ok);
  const apNodes = meshNodes.filter((n) => n.kind === 'ap');
  const piNode = meshNodes.find((n) => n.kind === 'pihole');
  const leaves = meshNodes.filter((n) => n.kind === 'device');

  // group leaves by the AP they're associated with (null = straight off router)
  const byAp = new Map(aps.map((a) => [a.id, []]));
  const routerKids = [];
  leaves.forEach((n) => (n.viaAp && byAp.has(n.viaAp) ? byAp.get(n.viaAp) : routerKids).push(n));

  // --- lay out AP hubs across a band beneath the router
  const apY = 372;
  const apCount = aps.length;
  const apSlot = W / (apCount + 1);
  const apPos = {};
  aps.forEach((a, i) => { apPos[a.id] = { x: apSlot * (i + 1), y: apY }; });

  const parts = [], nodeParts = [];
  // Gentle bezier: bows the link perpendicular to its run so dense fans stay readable.
  const link = (x1, y1, x2, y2, cls, id) => {
    const mx = (x1 + x2) / 2, my = (y1 + y2) / 2;
    const dx = x2 - x1, dy = y2 - y1, len = Math.hypot(dx, dy) || 1;
    const bow = Math.min(26, len * 0.13);
    const cx = mx - (dy / len) * bow, cy = my + (dx / len) * bow;
    parts.push(`<path class="mlink ${cls}"${id ? ` data-link="${id}"` : ''} d="M${x1.toFixed(1)},${y1.toFixed(1)} Q${cx.toFixed(1)},${cy.toFixed(1)} ${x2.toFixed(1)},${y2.toFixed(1)}"/>`);
  };

  // --- trunks
  link(router.x, router.y - 26, internet.x, internet.y + 22, d.internet?.up ? 'trunk' : 'trunk-down');
  link(router.x - 24, router.y, pihole.x + 24, pihole.y, d.pihole?.up ? 'trunk' : 'trunk-down');

  // --- fan children of a hub along an arc below it
  const fan = (kids, hub, radius, spreadDeg, baseDeg) => {
    const n = kids.length;
    kids.forEach((k, i) => {
      const t = n === 1 ? 0.5 : i / (n - 1);
      const ang = ((baseDeg - spreadDeg / 2 + t * spreadDeg) * Math.PI) / 180;
      k._x = hub.x + radius * Math.cos(ang);
      k._y = hub.y + radius * Math.sin(ang);
    });
  };

  // Router's own children arc out to the sides; radius grows with count so
  // labels don't collide on a busy LAN.
  const rKidR = 200 + Math.min(70, routerKids.length * 9);
  fan(routerKids, router, rKidR, Math.min(168, 60 + routerKids.length * 14), 90);
  // per-AP children fan out below each AP
  aps.forEach((a) => {
    const kids = byAp.get(a.id) || [];
    fan(kids, apPos[a.id], 120 + Math.min(60, kids.length * 11), Math.min(150, 50 + kids.length * 18), 90);
  });

  // --- links: router→AP, AP→device, router→device
  aps.forEach((a) => {
    const p = apPos[a.id];
    link(router.x, router.y + 26, p.x, p.y - 22, 'trunk', a.id);
    (byAp.get(a.id) || []).forEach((k) => {
      const cls = `${k.band === '5G' ? 'band5' : 'band24'} ${activityClass(k) === 'active' ? 'active' : ''}`;
      link(p.x, p.y, k._x, k._y, cls, a.id);
    });
  });
  routerKids.forEach((k) => link(router.x, router.y, k._x, k._y, `${k.direct ? 'direct' : 'indirect'} ${activityClass(k) === 'active' ? 'active' : ''}`));

  // --- hubs
  nodeParts.push(`<g class="mnode ${d.internet?.up ? 'hub' : 'hub hub-down'}">
    <circle class="body" cx="${internet.x}" cy="${internet.y}" r="22"/>
    ${iconSvg('globe', internet.x, internet.y, 22, 'var(--green)')}
    <text class="mlabel" x="${internet.x}" y="${internet.y - 32}">Internet</text>
    <text class="msub" x="${internet.x}" y="${internet.y - 19}">${d.internet?.up ? d.internet.latency + ' ms' : 'down'}</text></g>`);
  nodeParts.push(`<g class="mnode ${d.gateway?.up ? 'hub' : 'hub hub-down'}">
    ${d.gateway?.up ? `<circle class="pulse-ring" cx="${router.x}" cy="${router.y}" r="27"/>` : ''}
    <circle class="body" cx="${router.x}" cy="${router.y}" r="27"/>
    ${iconSvg('router', router.x, router.y, 26, 'var(--green)')}
    <text class="mlabel" x="${router.x + 46}" y="${router.y - 4}" style="text-anchor:start">${esc(d.gateway?.model || 'Router')}</text>
    <text class="msub" x="${router.x + 46}" y="${router.y + 9}" style="text-anchor:start">${esc(d.gateway?.host || '')}${d.gateway?.cpu != null ? ' · cpu ' + Math.round(d.gateway.cpu * 100) + '%' : ''}</text></g>`);
  nodeParts.push(`<g class="mnode pihole" data-pihole="1">
    ${d.pihole?.up ? `<circle class="pulse-ring" cx="${pihole.x}" cy="${pihole.y}" r="24"/>` : ''}
    <circle class="body" cx="${pihole.x}" cy="${pihole.y}" r="24"/>
    ${iconSvg('shield', pihole.x, pihole.y, 23, 'var(--cyan)')}
    <text class="mlabel" x="${pihole.x}" y="${pihole.y - 34}">Pi-hole</text>
    <text class="msub" x="${pihole.x}" y="${pihole.y - 21}">${esc(d.pihole?.host || '')}${d.pihole?.up ? ' · ' + d.pihole.latency + ' ms' : ''}</text></g>`);

  // --- AP hubs
  aps.forEach((a, i) => {
    const p = apPos[a.id];
    const kids = (byAp.get(a.id) || []).length;
    const bands = [a.wifi_2g ? '2.4G' : null, a.wifi_5g ? '5G' : null].filter(Boolean).join('+');
    nodeParts.push(`<g class="mnode ap" data-ap="${i}" data-apid="${a.id}">
      <circle class="pulse-ring" cx="${p.x}" cy="${p.y}" r="22"/>
      <circle class="body" cx="${p.x}" cy="${p.y}" r="22"/>
      ${iconSvg('ap', p.x, p.y, 22, 'var(--amber)')}
      <text class="mlabel" x="${p.x}" y="${p.y - 32}">${esc(a.label)}</text>
      <text class="msub" x="${p.x}" y="${p.y - 19}">${esc(a.ip)} · ${bands || 'radios off'} · ${kids} client${kids === 1 ? '' : 's'}</text></g>`);
  });

  // --- leaf devices
  const leafSvg = (n, idx) => {
    const cls = activityClass(n);
    const r = 11 + Math.min(11, Math.log2((n.numQueries || 0) + 1) * 1.3);
    const label = (n.name || n.ip || '?').slice(0, 15);
    const sub = n.signal != null ? `${n.signal} dBm` : (n.latency != null ? `${n.latency} ms` : (n.ip || ''));
    const key = deviceIconKey(n);
    const col = cls === 'active' ? 'var(--cyan)' : cls === 'idle' ? 'var(--violet)' : 'var(--dim)';
    return `<g class="mnode ${cls}" data-i="${idx}"${n.viaAp ? ` data-apid="${n.viaAp}"` : ''}>
      ${cls === 'active' ? `<circle class="pulse-ring" cx="${n._x.toFixed(1)}" cy="${n._y.toFixed(1)}" r="${r}"/>` : ''}
      <circle class="body" cx="${n._x.toFixed(1)}" cy="${n._y.toFixed(1)}" r="${r}"/>
      ${iconSvg(key, n._x, n._y, r * 1.15, col)}
      <text class="mlabel" x="${n._x.toFixed(1)}" y="${(n._y + r + 14).toFixed(1)}">${esc(label)}</text>
      <text class="msub" x="${n._x.toFixed(1)}" y="${(n._y + r + 26).toFixed(1)}">${esc(sub)}</text></g>`;
  };
  leaves.forEach((n) => { n._idx = meshNodes.indexOf(n); });
  leaves.forEach((n) => nodeParts.push(leafSvg(n, n._idx)));
  apNodes.forEach((n) => { n._x = null; }); // AP devices are drawn as hubs, not leaves

  svg.innerHTML = parts.join('') + nodeParts.join('');

  // --- interactivity
  const tip = $('#meshTip'), wrap = $('#meshWrap');
  const showTip = (e, html) => {
    const wr = wrap.getBoundingClientRect();
    tip.style.display = 'block';
    tip.style.left = Math.max(4, Math.min(e.clientX - wr.left + 16, wr.width - 275)) + 'px';
    tip.style.top = (e.clientY - wr.top + 14) + 'px';
    tip.innerHTML = html;
  };
  svg.querySelectorAll('.mnode[data-i]').forEach((g) => {
    const n = meshNodes[Number(g.dataset.i)];
    if (!n) return;
    g.addEventListener('mousemove', (e) => showTip(e, `<b>${esc(n.name || 'Unknown device')}</b><br>
      <span class="mono">${esc(n.ip)}${n.ips && n.ips.length > 1 ? ` (+${n.ips.length - 1})` : ''} · ${esc(n.mac || '')}</span><br>
      ${n.vendor ? esc(n.vendor) + '<br>' : ''}
      Path: ${n.viaAp ? `Wi-Fi <b>${esc(n.band || '')}</b> via <b>${esc(n.viaApLabel)}</b>` : n.direct ? 'wired, direct to router' : 'via switch'}<br>
      ${n.signal != null ? `Signal <b>${n.signal} dBm</b> · ` : ''}${n.latency != null ? `Latency <b>${n.latency} ms</b>` : 'latency —'}<br>
      Connected for ${fmtDur(Date.now() / 1000 - (n.firstSeen || Date.now() / 1000))} · last active ${n.lastQuery ? relTime(n.lastQuery) : '—'}<br>
      ${fmt(n.numQueries)} DNS queries · click to inspect`));
    g.addEventListener('mouseleave', () => { tip.style.display = 'none'; });
    g.addEventListener('click', () => { activateSection('queries'); $('#queryFilter').value = n.ip; loadQueries(); });
  });
  // Hovering an AP (or one of its clients) dims everything not on that path.
  const setFocus = (apId) => {
    svg.classList.toggle('focused', !!apId);
    svg.querySelectorAll('[data-apid]').forEach((el) => el.classList.toggle('lit', el.dataset.apid === apId));
  };
  svg.querySelectorAll('[data-apid]').forEach((el) => {
    el.addEventListener('mouseenter', () => setFocus(el.dataset.apid));
    el.addEventListener('mouseleave', () => setFocus(null));
  });

  svg.querySelectorAll('.mnode[data-ap]').forEach((g) => {
    const a = aps[Number(g.dataset.ap)];
    if (!a) return;
    // clicking an AP filters the device table to its clients
    g.addEventListener('click', () => {
      $('#meshFilter').value = a.label;
      renderMeshTable();
      $('#meshTable').closest('.glass').scrollIntoView({ behavior: 'smooth', block: 'start' });
      toast(`Filtered to ${a.label}`);
    });
    g.addEventListener('mousemove', (e) => showTip(e, `<b>${esc(a.label)}</b><br>
      <span class="mono">${esc(a.ip)}</span> · ${esc(a.model || '')}<br>
      ${a.firmware ? esc(String(a.firmware).split(' ').slice(0, 2).join(' ')) + '<br>' : ''}
      Radios: ${a.wifi_2g ? '2.4G ✓' : '2.4G ✕'} ${a.wifi_5g ? '5G ✓' : '5G ✕'}<br>
      ${a.wireless_clients} wireless client(s) · ${a.clients_total ?? '—'} total seen<br>
      ${a.cpu != null ? `CPU ${Math.round(a.cpu * 100)}%` : ''} ${a.mem != null ? `· RAM ${Math.round(a.mem * 100)}%` : ''}
      ${a.latency != null ? `<br>Latency <b>${a.latency} ms</b>` : ''}`));
    g.addEventListener('mouseleave', () => { tip.style.display = 'none'; });
  });
  svg.querySelector('[data-pihole]')?.addEventListener('click', () => activateSection('diagnostics'));

  // --- icon legend (only the types actually present)
  const present = [...new Set(leaves.map(deviceIconKey))].sort();
  $('#meshLegend').innerHTML = present.map((k) =>
    `<span class="lg-ico">${iconHtml(k, 13)}${esc(ICON_LABEL[k] || k)}</span>`).join('');
}
